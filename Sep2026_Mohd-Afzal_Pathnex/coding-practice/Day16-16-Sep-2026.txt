Ansible — Create and Start EC2 Instances

- name: Create EC2 instances
  hosts: localhost
  become: yes
  tasks:
    - name: Launch EC2 instance in us-east-1
      ec2_instance:
        name: afzal-ec2-us-east-1
        key_name: afzal-key
        region: us-east-1
        instance_type: t2.micro
        image: ami-0abcd1234abcd1234
        wait: yes
    - name: Launch EC2 instance in us-west-1
      ec2_instance:
        name: afzal-ec2-us-west-1
        key_name: afzal-key
        region: us-west-1
        instance_type: t2.micro
        image: ami-0abcd1234abcd1234
        wait: yes

resource "aws_vpc" "us_east_vpc" {

Kubernetes — Multi-Cluster Deployment with Federation

apiVersion: apps/v1
kind: Deployment
metadata:
  name: afzal-deployment
  labels:
    app: afzal-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: afzal-app
  template:
    metadata:
      labels:
        app: afzal-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80

---

apiVersion: v1
kind: Service
metadata:
  name: afzal-service
spec:
  selector:
    app: afzal-app
  ports:
    - protocol: TCP
      port: 80

Jenkinsfile — Deploy to Multi-Cluster Environments

pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                docker.build('afzal-web-app')
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    if (env.CLUSTER == 'us-east') {
                        sh 'kubectl --context=us-east apply -f kubernetes/deployment.yaml'
                    } else {
                        sh 'kubectl --context=us-west apply -f kubernetes/deployment.yaml'
                    }
                }
            }
        }
    }
}

GitLab CI/CD — Multi-Region Deployment with Kubernetes

stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t afzal-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push afzal-web-app

deploy:
  stage: deploy
  script:
    - |
      if [ "$CI_COMMIT_REF_NAME" == "main" ]; then
        kubectl --context=us-east apply -f kubernetes/prod-deployment.yaml
      else
        kubectl --context=us-west apply -f kubernetes/dev-deployment.yaml
      fi
