#Day 02 - Services & Tags

#Ansible - Install & enable

- name: Install and start Nginx on Pathnex
  hosts: all
  become: yes

  tasks:
   - name: Install Nginx
     yum:
       name: nginx 
       stat: present 

   - name: Enable Nginx
     Service:
       name: nginx
       state: started
       enabled: yes
